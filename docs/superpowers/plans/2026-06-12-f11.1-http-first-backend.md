# F11.1 — Backend HTTP-first Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Todas as APIs hoje Wails-bound passam a ser servidas por HTTP+SSE no `internal/localhttp` (com bearer token), e o frontend React passa a consumi-las por `fetch()`/`EventSource` — com o Wails ainda funcionando como janela (F11.2 troca a casca).

**Architecture:** Novo pacote `internal/httpapi` com handlers promovidos do template e2e (`internal/api/http_e2e.go`), montados em `/api/*` no servidor `localhttp` que já existe em produção. Eventos via `SSEEmitter` (implementa `events.Emitter`) em `/api/events`, emitindo em paralelo com o `WailsEmitter` via `MultiEmitter`. No frontend, `lib/api.ts` mantém a superfície pública (`api.*`) trocando bindings por HTTP; config (porta+token) resolvida por query params (browser/e2e) ou binding temporário `GetServerConfig()` (janela Wails).

**Tech Stack:** Go 1.22+ (`net/http` com method patterns, `httptest`), SSE (`text/event-stream`), React/TS (fetch, EventSource), vitest + MSW, Playwright.

**Spec:** `docs/superpowers/specs/2026-06-12-f11-electron-http-first-design.md`

**Regras do repo que valem aqui:** TDD iron-law (ADR-0004) — teste primeiro em todo código novo; commits frequentes direto na `main`; `make` targets existentes (`go test ./...`, `cd ui && pnpm test`).

- Estilo de rota: RPC `POST /api/<área>/<ação>` com os mesmos shapes JSON do template e2e (mudança mínima, template provado). REST fica para nunca — YAGNI.

```go
// Package httpapi serves the app APIs over loopback HTTP. F11.1: promoted
// from the e2e_http template so production, dev and Playwright share one
// transport (spec: 2026-06-12-f11-electron-http-first-design.md).
package httpapi

import (
	"crypto/subtle"
	"net/http"
)

// RequireToken guards routes with the boot-generated app token. Accepts
// `Authorization: Bearer <t>` or `?token=<t>` — EventSource cannot set
// headers, so SSE authenticates via query param. An empty configured
// token rejects everything (fail-closed if boot wiring breaks).
func RequireToken(token string, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == http.MethodOptions { // CORS preflight carries no auth
			next.ServeHTTP(w, r)
			return
		}
		authOK := subtle.ConstantTimeCompare([]byte(r.Header.Get("Authorization")), []byte("Bearer "+token)) == 1
		queryOK := subtle.ConstantTimeCompare([]byte(r.URL.Query().Get("token")), []byte(token)) == 1
		if token != "" && (authOK || queryOK) {
			next.ServeHTTP(w, r)
			return
		}
		// http.Error would reset Content-Type to text/plain — write the 401 by hand.
		w.Header().Set("Content-Type", "application/json")
		w.Header().Set("X-Content-Type-Options", "nosniff")
		w.WriteHeader(http.StatusUnauthorized)
		_, _ = w.Write([]byte(`{"error":"unauthorized"}`))
	})
}
```

```go
package events

import (
	"encoding/json"
	"net/http"
	"sync"
)

type sseEnvelope struct {
	Name    string `json:"name"`
	Payload any    `json:"payload"`
}

// SSEEmitter broadcasts bus events to connected /api/events clients.
// Implements Emitter. A client whose buffer fills (slow consumer) is
// dropped rather than blocking Emit — producers run on hot paths
// (PTY output, hook handlers). EventSource auto-reconnects, so a
// dropped client recovers on its own and React Query re-syncs state.
type SSEEmitter struct {
	mu      sync.Mutex
	clients map[chan []byte]struct{}
}

func NewSSEEmitter() *SSEEmitter {
	return &SSEEmitter{clients: map[chan []byte]struct{}{}}
}

func (e *SSEEmitter) Emit(name string, payload any) {
	data, err := json.Marshal(sseEnvelope{Name: name, Payload: payload})
	if err != nil {
		return
	}
	e.mu.Lock()
	defer e.mu.Unlock()
	for ch := range e.clients {
		select {
		case ch <- data:
		default: // full buffer = slow consumer; drop it
			delete(e.clients, ch)
			close(ch)
		}
	}
}

// ClientCount exists for tests and the health snapshot; not load-bearing.
func (e *SSEEmitter) ClientCount() int {
	e.mu.Lock()
	defer e.mu.Unlock()
	return len(e.clients)
}

func (e *SSEEmitter) subscribe() chan []byte {
	ch := make(chan []byte, 1024)
	e.mu.Lock()
	e.clients[ch] = struct{}{}
	e.mu.Unlock()
	return ch
}

func (e *SSEEmitter) unsubscribe(ch chan []byte) {
	e.mu.Lock()
	defer e.mu.Unlock()
	if _, ok := e.clients[ch]; ok {
		delete(e.clients, ch)
		close(ch)
	}
}

func (e *SSEEmitter) Handler() http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		fl, ok := w.(http.Flusher)
		if !ok {
			http.Error(w, "streaming unsupported", http.StatusInternalServerError)
			return
		}
		w.Header().Set("Content-Type", "text/event-stream")
		w.Header().Set("Cache-Control", "no-cache")
		ch := e.subscribe()
		defer e.unsubscribe(ch)
		w.WriteHeader(http.StatusOK)
		fl.Flush()
		for {
			select {
			case <-r.Context().Done():
				return
			case data, open := <-ch:
				if !open { // dropped by Emit (slow consumer)
					return
				}
				_, _ = w.Write([]byte("data: "))
				_, _ = w.Write(data)
				_, _ = w.Write([]byte("\n\n"))
				fl.Flush()
			}
		}
	})
}
```

```go
package httpapi

import (
	"encoding/json"
	"net/http"

	"github.com/marcosdid/jarvis/internal/api"
)

// Deps carries every Wails-bound API the HTTP layer exposes. Nil members
// are allowed in tests; production wires all of them (main.go).
type Deps struct {
	Token     string
	Tasks     *api.TasksAPI
	Projects  *api.ProjectsAPI
	Worktrees *api.WorktreesAPI
	Sessions  *api.SessionsAPI
	Master    *api.MasterAPI
	Runs      *api.RunsAPI
	Bootstrap *api.BootstrapAPI
	Catalog   *api.CatalogAPI
	Health    *api.HealthAPI
	Events    http.Handler // SSEEmitter.Handler() — servido DENTRO do mux para herdar CORS+token
}

type Server struct{ deps Deps }

func New(deps Deps) *Server { return &Server{deps: deps} }

// Handler returns the authenticated /api/* mux. CORS mirrors the e2e
// posture (loopback-only listener + token; origin check adds nothing)
// and MUST allow the Authorization header for vite dev/preview.
func (s *Server) Handler() http.Handler {
	mux := http.NewServeMux()
	s.mount(mux)
	return withCORS(RequireToken(s.deps.Token, mux))
}

func withCORS(h http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Access-Control-Allow-Origin", "*")
		w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
		w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		h.ServeHTTP(w, r)
	})
}

func writeJSON(w http.ResponseWriter, v any) {
	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(v)
}

func writeErr(w http.ResponseWriter, status int, err error) {
	w.WriteHeader(status)
	writeJSON(w, map[string]string{"error": err.Error()})
}

func (s *Server) mount(mux *http.ServeMux) {
	// ... rotas — ver tabela abaixo
	// ⚠️ Events é registrado direto no mux — ServeMux.Handle(nil) PANICA.
	// Guard obrigatório (os testes constroem Deps sem Events):
	if s.deps.Events != nil {
		mux.Handle("GET /api/events", s.deps.Events)
	}
}
```

```go
package httpapi

import (
	"io/fs"
	"net/http"
	"path"
	"strings"
)

// SPAHandler serves the embedded ui/dist with index.html fallback so
// client-side routes deep-link correctly. F11.1: mounted but the Wails
// asset server still owns the production window; F11.2 makes this the
// only UI origin.
func SPAHandler(dist fs.FS) http.Handler {
	fileServer := http.FileServerFS(dist)
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		p := strings.TrimPrefix(path.Clean(r.URL.Path), "/")
		if p != "" {
			if _, err := fs.Stat(dist, p); err != nil {
				r.URL.Path = "/" // SPA fallback
			}
		}
		fileServer.ServeHTTP(w, r)
	})
}
```

```ts
// Resolves where the Go backend lives. Sources, in order:
// 1. `?apiPort=N&apiToken=T` — browser dev & Playwright e2e.
// 2. Temporary Wails binding GetServerConfig() — production window (F11.1 only;
//    F11.2's Electron shell injects query params instead and the binding dies).
export type ServerConfig = { baseUrl: string; token: string };

let cached: ServerConfig | null = null;

export async function getServerConfig(): Promise<ServerConfig> {
  if (cached) return cached;
  const params = new URLSearchParams(window.location.search);
  const port = params.get('apiPort');
  if (port) {
    cached = { baseUrl: `http://127.0.0.1:${port}`, token: params.get('apiToken') ?? '' };
    return cached;
  }
  const binding = (window as unknown as {
    go?: { main?: { App?: { GetServerConfig?: () => Promise<{ base_url: string; token: string }> } } };
  }).go?.main?.App?.GetServerConfig;
  if (binding) {
    const cfg = await binding();
    cached = { baseUrl: cfg.base_url, token: cfg.token };
    return cached;
  }
  throw new Error('server config unavailable: no apiPort param and no Wails binding');
}

export function __resetServerConfigForTests(): void {
  cached = null;
}
```

```ts
// Singleton EventSource over /api/events. All bus events arrive as
// `data: {"name": "...", "payload": {...}}` envelopes (default message
// event — names are dynamic, addEventListener-per-name doesn't fit).
import { sseURL } from './http';

type Listener = (payload: unknown) => void;
export type StreamState = 'connected' | 'reconnecting' | 'offline';
type StateListener = (s: StreamState) => void;

const listeners = new Map<string, Set<Listener>>();
const stateListeners = new Set<StateListener>();
let source: EventSource | null = null;

export async function ensureEventStream(): Promise<void> {
  if (source) return;
  const url = await sseURL('/api/events');
  source = new EventSource(url);
  source.onopen = () => stateListeners.forEach((cb) => cb('connected'));
  // EventSource auto-reconnects; surface the gap so the ErrorBanner shows.
  source.onerror = () => stateListeners.forEach((cb) => cb('reconnecting'));
  source.onmessage = (ev) => {
    try {
      const { name, payload } = JSON.parse(ev.data) as { name: string; payload: unknown };
      listeners.get(name)?.forEach((cb) => cb(payload));
    } catch {
      // malformed frame: ignore — next frame resyncs
    }
  };
}

export function onEvent(name: string, cb: Listener): () => void {
  const set = listeners.get(name) ?? new Set();
  set.add(cb);
  listeners.set(name, set);
  return () => {
    set.delete(cb);
  };
}

export function onStreamState(cb: StateListener): () => void {
  stateListeners.add(cb);
  return () => stateListeners.delete(cb);
}

export function closeEventStream(): void {
  source?.close();
  source = null;
  stateListeners.forEach((cb) => cb('offline'));
}
```
